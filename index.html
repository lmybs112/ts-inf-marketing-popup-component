複製以下代碼到您的網站：

<script>
function init() {
    // 檢查是否已經初始化過，如果是則先清除
    const existingScript = document.querySelector('script[src="./inf-marketing-popup-component.js"]');
    if (existingScript) {
        existingScript.remove();
    }
    
    // 清除已存在的彈窗組件
    const existingPopups = document.querySelectorAll('inf-marketing-popup-component');
    existingPopups.forEach(popup => popup.remove());
    
    const script = document.createElement('script');
    script.src = 'https://ts-inf-marketing-popup-component.vercel.app/inf-marketing-popup-component.js';
    script.sync = true;
    script.onload = () => {
        window.createInfMarketingPopupByBrand('ALMI');
    };
    document.body.appendChild(script);
}

// 如果 DOM 還在 loading，就等事件；否則直接執行
if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
} else {
    init();
}
</script>
